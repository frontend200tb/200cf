Задача A. Шлёпки

Решение 
#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);

  int t; // число тестов
  cin >> t;

  while (t--) {
    int n; // число монстров
    long long c; // наша боевая мощь
    int k; // число шлепок
    cin >> n >> c >> k;

    vector<int> A(n); // боевая мощь монстров
    for (int i = 0; i < n; i++) {
      cin >> A[i];
    }

    sort(A.begin(), A.end());

    int next = 0; // следующий монстр
    while (k > 0 && next < n) {
      if (c > A[next]) {
        int dif = c - A[next];
        if (k > dif) {
          c += dif + A[next];
          k -= dif;
        } else {
          c += k + A[next];
          k = 0;
        }
        next++;
      } else if (c == A[next]) {
        c += A[next];
        next++;
      } else {
        break;
      }
    }

    while (next < n) {
      if (c >= A[next]) {
        c += A[next];
        next++;
      } else {
        break;
      }
    }

    cout << c << '\n';
  }
}

Задача B. Массив

Решение 
#include <iostream>
#include <vector>

using namespace std;

int main() {
	int t; // число тестов
	cin >> t;

	while (t--) {
		int n; // длина массива
		cin >> n;

		vector<int> A(n);
		for (int i = 0; i < n; i++) {
			cin >> A[i];
		}

		vector<int> B(n);
		vector<int> C(n);
		for (int i = n - 1; i >= 0; i--) {
			B[i] = C[i] = 0;
			for (int j = i + 1; j < n; j++) {
				B[i] += (A[i] > A[j]);
				C[i] += (A[i] < A[j]);
			}
			B[i] = max(B[i], C[i]);
		}

		for (int i = 0; i < n - 1; i++) {
			cout << B[i] << ' ';
		}
		cout << B[n - 1] << '\n';
	}
}

Задача C. Найдите ноль

Решение

В массиве длины 2n есть n нулей и n различных ненулевых чисел (от 1 до n). Если два элемента равны, то они обязательно оба нули (так как ненулевые числа уникальны).

Следовательно, если мы найдём любую пару индексов (i, j), для которой query(i, j) = 1, то оба элемента — нули, и мы можем вывести любой из них.

Алгоритм гарантированно находит ноль за n+1 запросов, используя простую логику проверки пар и небольшой fallback для первых двух элементов.

#include <iostream>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    int ans = -1;

    // проверка пар среди первых 2n-2 элементов
    for (int i = 1; i < n; ++i) {
      cout << "? " << 2 * i + 1
      << " " << 2 * i + 2 << endl;
      int resp;
      cin >> resp;
      if (resp == 1) {
        ans = 2 * i + 1;
        break;
      }
    }

    // обработка случая, когда не нашли пару нулей
    if (ans == -1) {
      cout << "? 1 3" << endl;
      int resp;
      cin >> resp;
      if (resp == 1) {
        ans = 1;
      } else {
        cout << "? 1 4" << endl;
        cin >> resp;
        if (resp == 1) {
          ans = 1;
        } else {
          ans = 2;
        }
      }
    }

    cout << "! " << ans << endl;
  }
}

Задача E. Тривиальная задача на строки

Решение 
#include <iostream>
#include <string>
#include <vector>

using namespace std;

long long work(const string& s) {
  int n = s.length();
  vector<int> p(n, 0);
  vector<int> b(n, 0);

  // Построение префикс-функции
  for (int i = 1; i < n; i++) {
    int j = p[i - 1];
    while (j > 0 && s[i] != s[j]) {
      j = p[j - 1];
    }
    if (s[i] == s[j]) {
      j++;
    }
    p[i] = j;

    if (j == 0) {
      b[i] = 0;
    } else if (p[j - 1] == 0) {
      b[i] = j;
    } else {
      b[i] = b[j - 1];
    }
  }

  // Подсчет ответа - используем long long для предотвращения переполнения
  long long ans = 0;
  vector<long long> dp(n, 0);

  for (int i = 0; i < n; i++) {
    if (b[i] == 0) {
      dp[i] = 1;
    } else {
      int prev_index = i - b[i];
      if (prev_index >= 0) {
        dp[i] = dp[prev_index] + 1;
      } else {
        dp[i] = 1;
      }
    }
    ans += dp[i];
  }

  return ans;
}

int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);

  int t;
  cin >> t;

  while (t--) {
    int n, q;
    cin >> n >> q;
    string s;
    cin >> s;

    while (q--) {
      int l, r;
      cin >> l >> r;
      cout << work(s.substr(l - 1, r - l + 1)) << '\n';
    }
  }

  return 0;
}

Задача F. Динамические значения и максимальный счёт

Решение

Алгоритм находит максимальный возможный счёт после выполнения k операций, где каждая операция:

  1. Выбирает корень дерева
  2. Добавляет значение корня к счёту и обнуляет его
  3. Перераспределяет значения всех не-листовых вершин к листьям в их поддеревьях
#include <iostream>
#include <vector>
#include <set>
#include <algorithm>
#include <cstdio>

using namespace std;

const int N = 300009;
int n, k, t, a[N], tt[N];
long long b[N], ans[N], sum, mxa;

// FLONG используется в множестве для хранения вершин
// с их накопленными значениями, отсортированных по убыванию.
struct FLONG {
  long long val; // накопленное значение в вершине
  int nd; // номер вершины
  FLONG(long long v = 0, int d = 0) : val(v), nd(d) {}

  // Сортировка: сначала по убыванию val, при равенстве по возрастанию nd
  bool operator<(const FLONG& b) const {
    return val > b.val || (val == b.val && nd < b.nd);
  }
};

set<FLONG> s;
set<FLONG>::iterator o;

// pir хранит информацию о самом дальнем листе:
// to - индекс листа
// len - расстояние до него
// Сортировка : сначала по убыванию длины, при равенстве - по возрастанию индекса
// mn[v] - самый дальний лист в поддереве v
// se[v] - второй по дальности лист
// lf[v] - самый дальний лист вне поддерева v
struct pir {
  int to, len; // to - вершина-лист, len - расстояние до неё
  pir(int t = 0, int l = 0) : to(t), len(l) {}

  bool operator<(const pir& b) const {
    return len > b.len || (len == b.len && to < b.to);
  }
} mn[N], se[N], lf[N];

vector<int> e[N];

// Первый обход дерева (srh)
// Для каждой вершины находит два самых удалённых листа
// в её поддереве. Это нужно для быстрого определения,
// куда будет перемещаться значение вершины при разных корнях.
void srh(int v, int fa) {
  mn[v] = pir(v, 0); // для листа - он сам
  se[v] = pir(0, ~0x3f3f3f3f); // очень маленькое значение
  lf[v] = pir(0, ~0x3f3f3f3f);

  for (unsigned i = 0; i < e[v].size(); i++) {
    if (e[v][i] != fa) {
      srh(e[v][i], v);
      pir th = mn[e[v][i]];
      th.len++; // увеличиваем расстояние на 1 (ребро до дочерней вершины)
      // Обновляем два самых дальних листа
      if (th < se[v])
        swap(th, se[v]);
      if (se[v] < mn[v])
        swap(se[v], mn[v]);
    }
  }
}

// Инициализация для корня 1 (gt1)
// Вычисляет счёт для случая, когда корень - вершина 1:
// Добавляет значение корня
// Определяет, куда перераспределятся значения остальных вершин
// Выбирает k - 1 наибольших накопленных значений(плюс корень даёт всего k операций)
void gt1() {
  sum = ans[1] = 0;
  ans[1] += a[1]; // добавляем значение корня

  // Для каждой не-корневой вершины i, её значение пойдёт к mn[i].to
  for (int i = 2; i <= n; i++) {
    b[mn[i].to] += a[i];
    tt[i] = mn[i].to; // запоминаем, куда пойдёт значение
  }

  // Собираем все накопленные значения в множество
  s.clear();
  for (int i = 2; i <= n; i++) {
    s.insert(FLONG(b[i], i));
  }

  // Берём k-1 самых больших значений (k-1, потому что корень уже взят)
  o = s.begin();
  ans[1] += o->val;
  sum += o->val;

  for (int i = 1; i < k - 1; i++) {
    o++;
    ans[1] += o->val;
    sum += o->val;
  }

  s.insert(FLONG(-1ll, 0));
  s.insert(FLONG(-2ll, 0));
}

// Функции для динамического обновления (era, ins, stp, sett)

// era(x) - удаляет вершину x из множества
void era(int x) {
  FLONG u = FLONG(b[x], x);
  if (!(*o < u)) {
    o++;
    sum += o->val;
    sum -= b[x];
  }
  s.erase(u);
}

// ins(x) - добавляет вершину x в множество
void ins(int x) {
  FLONG u = FLONG(b[x], x);
  s.insert(u);
  if (!(*o < u)) {
    sum -= o->val;
    sum += b[x];
    o--;
  }
}

// stp(x) - перемещает значение вершины x
// (удаляет из старого места, добавляет в новое)
void stp(int x) {
  era(tt[x]);
  b[tt[x]] -= a[x];
  ins(tt[x]);
  tt[x] = 0;
}

// sett(x, ttp) - устанавливает новое назначение для вершины x
void sett(int x, int ttp) {
  tt[x] = ttp;
  era(tt[x]);
  b[tt[x]] += a[x];
  ins(tt[x]);
}

// Второй обход для пересчёта всех корней (srho)
// Обходит дерево, пересчитывая ответ для каждой вершины
// как потенциального корня.При переходе к новому корню :
// Обновляет информацию о том, куда перераспределяются значения
// Поддерживает актуальное множество накопленных значений
// Использует технику "переподвешивания" дерева для эффективного пересчёта
void srho(int v, int fa) {
  if (fa != 0) {
    stp(v); // убираем v из старого назначения
    era(v); // удаляем v из множества
    ins(fa); // добавляем fa в множество
    sett(fa, lf[fa].to); // обновляем назначение для fa
    ans[v] = sum + a[v]; // считаем ответ для корня v
  }

  // Рекурсивно обрабатываем детей
  for (unsigned i = 0; i < e[v].size(); i++) {
    if (e[v][i] != fa) {
      pir u = lf[fa];
      u.len++;
      // Вычисляем самый дальний лист вне поддерева v
      if (mn[v].to == mn[e[v][i]].to)
        lf[v] = min(u, se[v]);
      else
        lf[v] = min(u, mn[v]);
      srho(e[v][i], v);
    }
  }

  // Откатываем изменения при возврате
  if (fa != 0) {
    ins(v);
    sett(v, mn[v].to);
    stp(fa);
    era(fa);
  }
}

int main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);

  cin >> t;

  while (t--) {
    cin >> n >> k;
    mxa = 0;

    for (int i = 1; i <= n; i++) {
      cin >> a[i];
      mxa = max(mxa, 1ll * a[i]);
      e[i].clear();
      b[i] = 0;
    }

    for (int i = 1; i < n; i++) {
      int u, v;
      cin >> u >> v;
      e[u].push_back(v);
      e[v].push_back(u);
    }

    if (k == 1) {
      cout << mxa << "\n";
      continue;
    }

    lf[0].len = ~0x3f3f3f3f;
    srh(1, 0);
    gt1();
    srho(1, 0);

    for (int i = 1; i <= n; i++) {
      mxa = max(mxa, ans[i]);
    }

    cout << mxa << "\n";
  }

  return 0;
}
