Задача A. Шлёпки

Решение 
#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);

  int t; // число тестов
  cin >> t;

  while (t--) {
    int n; // число монстров
    long long c; // наша боевая мощь
    int k; // число шлепок
    cin >> n >> c >> k;

    vector<int> A(n); // боевая мощь монстров
    for (int i = 0; i < n; i++) {
      cin >> A[i];
    }

    sort(A.begin(), A.end());

    int next = 0; // следующий монстр
    while (k > 0 && next < n) {
      if (c > A[next]) {
        int dif = c - A[next];
        if (k > dif) {
          c += dif + A[next];
          k -= dif;
        } else {
          c += k + A[next];
          k = 0;
        }
        next++;
      } else if (c == A[next]) {
        c += A[next];
        next++;
      } else {
        break;
      }
    }

    while (next < n) {
      if (c >= A[next]) {
        c += A[next];
        next++;
      } else {
        break;
      }
    }

    cout << c << '\n';
  }
}

Задача B. Массив

Решение 
#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> a(n);
        for (int i = 0; i < n; i++) {
            cin >> a[i];
        }

        vector<int> ans(n);

        for (int i = 0; i < n; i++) {
            vector<pair<double, int> > events;

            int cnt_left = 0; // количество a_j < a_i (побеждают при k -> -inf)
            int cnt_right = 0; // количество a_j > a_i (побеждают при k -> +inf)

            for (int j = i + 1; j < n; j++) {
                if (a[j] > a[i]) {
                    cnt_right++;
                    double mid = (a[i] + a[j]) / 2.0;
                    events.emplace_back(mid, +1); // при k > mid, этот j начинает побеждать
                } else if (a[j] < a[i]) {
                    cnt_left++;
                    double mid = (a[i] + a[j]) / 2.0;
                    events.emplace_back(mid, -1); // при k > mid, этот j перестает побеждать
                }
            }

            // Начальное состояние: при k -> -inf побеждают все a_j < a_i
            int current = cnt_left;
            int best = max(current, cnt_right);

            sort(events.begin(), events.end());

            for (size_t idx = 0; idx < events.size(); ) {
                double pos = events[idx].first;
                int delta = 0;
                while (idx < events.size() && events[idx].first == pos) {
                    delta += events[idx].second;
                    idx++;
                }
                current += delta;
                best = max(best, current);
            }

            ans[i] = best;
        }

        for (int i = 0; i < n; i++) {
            cout << ans[i];
            if (i < n - 1) cout << " ";
        }
        cout << "\n";
    }

    return 0;
}
