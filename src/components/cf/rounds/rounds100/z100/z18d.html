В прошлом году Вася подрабатывал продажей компьютерных флешек. В каждый из n дней его работы происходило одно из двух:

К Васе приходил покупатель и спрашивал флешку на 2x мегабайт. Если у Васи была такая флешка, он продавал ее и получал 2x бурлей. Вася выигрывал какую-нибудь олимпиаду по программированию и получал в качестве приза флешку на 2x мегабайт. При этом он мог выбрать, подарить ли ему эту флешку кому-то из друзей, или оставить ее себе. Вася никогда не хранил у себя больше одной флешки, так как боялся перепутать их объемы и случайно кого-нибудь обмануть. Также известно, что для каждого объема флешки было не более одного покупателя, желающего купить такую флешку. Сейчас, зная все запросы покупателей и все призы на олимпиадах по программированию за прошедшие n дней, Васе интересно, сколько денег он мог бы заработать, если бы действовал оптимально.

Входные данные

В первой строке входных данных содержится число n (1 ≤ n ≤ 5000) — сколько дней работал Вася. Следующие n строк содержат описание дней. Строка вида sell x описывает день, в который к Васе приходил покупатель за флешкой на 2x мегабайт (0 ≤ x ≤ 2000). Гарантируется, что для каждого x задано не более одной строки вида sell x. Строка вида win x описывает день, в который Вася выиграл флешку на 2x мегабайт (0 ≤ x ≤ 2000).

Выходные данные

Выведите наибольший возможный заработок Васи в бурлях, если бы он заранее знал все события. Не забудьте, что Вася не может хранить у себя одновременно больше одной флешки.

Решение
Задача 18D
Codeforces Beta Round 18 (Div.2) 2010.06.16

У Васи может быть только одна флешка. Каждый день происходит либо win x (выиграл флешку объема 2^x), либо sell x (пришел покупатель за флешкой 2^x). Нужно максимизировать заработок.

Алгоритм использует жадный подход с выбором продаж в порядке убывания объема.

Длинная арифметика для подсчета суммы. Так как 2^2000 — это огромное число (около 600 десятичных цифр), используем массив для хранения числа в десятичной системе.

#include 
#include 
using namespace std;

const int MAX_DIGITS = 610;

int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);

  int days;
  cin >> days;

  int eventType[5000];
  int volume[5000];
  int wasSold[2001] = { 0 };

  // result[1] - единицы, result[2] - десятки, ...
  int result[MAX_DIGITS] = { 0 };

  string operation;
  int x;

  for (int i = 0; i < days; i++) {
    cin >> operation >> x;
    volume[i] = x;
    if (operation[0] == 'w')
      eventType[i] = 0;
    else
      eventType[i] = 1;
  }

  while (true) {
    // 1. Находим sell с максимальным объемом
    int maxVolume = -1;
    int sellPos = -1;

    for (int i = 0; i < days; i++) {
      if (eventType[i] == 1 && maxVolume < volume[i]) {
        maxVolume = volume[i];
        sellPos = i;
      }
    }

    if (maxVolume < 0) {
      break;
    }

    // 2. Ищем win того же объема до sell
    int winPos;
    for (winPos = sellPos - 1; winPos >= 0 && eventType[winPos] >= 0; winPos--) {
      if (eventType[winPos] == 0 && volume[winPos] == maxVolume)
        break;
    }

    // 3. Если win не найден - помечаем sell как невозможный
    if (winPos < 0 || eventType[winPos] < 0) {
      eventType[sellPos] = 2;
      continue;
    }

    // 4. Нашли пару - запоминаем и удаляем промежуточные события
    wasSold[maxVolume] = 1;

    for (int i = winPos; i <= sellPos; i++) {
      eventType[i] = -1;
    }
  }

  for (int i = 0; i < MAX_DIGITS; i++) {
    result[i] = 0;
  }

  for (int x = 2000; x >= 0; x--) {
    // Умножаем текущее число на 2
    for (int i = 1; i < MAX_DIGITS; i++) {
      result[i] <<= 1; // умножаем разряд на 2
      if (result[i - 1] > 9) {
        result[i - 1] -= 10;
        result[i]++;
      }
    }
    // Добавляем 1 (прибавляем 2^x)
    result[1] += wasSold[x];
  }

  int pos = MAX_DIGITS - 1;
  while (pos > 1 && result[pos] == 0) pos--;

  for (int i = pos; i >= 1; i--) {
    cout << result[i];
  }
  cout << '\n';

  return 0;
}
