Дан циклический массив a0, a1, ..., an - 1. Есть два типа операций:

  • inc(lf, rg, v) — увеличить каждый элемент на отрезке [lf, rg] (включительно) на v;
  • rmq(lf, rg) — найти минимальное значение на отрезке [lf, rg] (включительно).

Мы считаем, что все отрезки — циклические, например если n = 5 и lf = 3, rg = 1, то имеется в виду последовательность индексов: 3, 4, 0, 1.

Напишите программу, которая будет выполнять заданную последовательность операций.

Входные данные

Первая строка содержит целое число n (1 ≤ n ≤ 200000). Следующая строка описывает начальное состояние массива: a0, a1, ..., an - 1 ( - 10⁶ ≤ ai ≤ 10⁶), ai целые. Третья строка содержит целое число m (0 ≤ m ≤ 200000), m — количество операций. Следующие m строк описывают операции. Если строка содержит два целых числа lf, rg (0 ≤ lf, rg ≤ n - 1), то это она задает операцию rmq, а если она содержит три целых числа lf, rg, v (0 ≤ lf, rg ≤ n - 1; - 10⁶ ≤ v ≤ 10⁶) — операцию inc.

Выходные данные

Для каждой операции rmq выведите ее результат. Пожалуйста, не используйте спецификатор %lld для чтения или записи 64-х битовых чисел на С++. Рекомендуется использовать поток cout (также вы можете использовать спецификатор %I64d).

#include 

using namespace std;

const int buk = 450;
int n, m, q;
long long a[200005], add[200005], b[200005];

void update(int cur) {
  b[cur] = 1e18;
  for (int i = cur * buk; i < cur * buk + buk; i++) {
    b[cur] = min(b[cur], a[i]);
  }
}

long long query(int l, int r) {
  long long res = 1e18;
  int i = l;
  for (;i % buk != 0 && i <= r;++i) {
    res = min(res, a[i] + add[i / buk]);
  }
  for (;i + buk - 1 <= r;i += buk) {
    res = min(res, b[i / buk] + add[i / buk]);
  }
  for (;i <= r;++i) {
    res = min(res, a[i] + add[i / buk]);
  }
  return res;
}
void inc(int l, int r, int d) {
  int i = l;
  while (i % buk != 0 && i <= r) {
    a[i++] += d;
  }
  update(l / buk);
  for (;i + buk - 1 <= r;i += buk) {
    add[i / buk] += d;
  }
  while (i <= r) {
    a[i++] += d;
  }
  update(r / buk);
}

int main() {
  ios::sync_with_stdio(false);
  cin >> n;

  for (int i = 0;i < n;i++) {
    cin >> a[i];
  }

  for (int i = 0;i <= (n - 1) / buk;i++) {
    update(i);
  }

  cin >> q;
  while (q--)	{
    int l, r, d;
    cin >> l >> r;
    if (cin.get() == 10) {
      if (l <= r) {
        cout << query(l, r) << endl;
      } else {
        cout << min(query(0, r), query(l, n - 1)) << endl;
      }
    } else {
      cin >> d;
      if (l <= r) {
        inc(l, r, d);
      } else {
        inc(0, r, d);
        inc(l, n - 1, d);
      }
    }
  }

  return 0;
}
