Недавно Вася придумал новую игру с деревом (напоминаем, что дерево — это связный граф без циклов): он удаляет любое (возможно, нулевое) количество ребер данного дерева, и подсчитывает произведение размеров получившихся компонент связности. Ваша задача — для заданного дерева определить, какое наибольшее число сможет получить Вася в своей новой игре.

Входные данные

В первой строке содержится целое число n (1 ≤ n ≤ 700) — число вершин в дереве. Далее в n - 1 строках записаны ребра дерева. Каждая строка содержит пару номеров вершин, соединенных ребром, ai, bi (1 ≤ ai, bi ≤ n). Гарантируется, описанный во входных данных граф является деревом.

Выходные данные

Выведите единственное число — какое наибольшее произведение размеров компонент связности можно получить, удалив из дерева некоторые ребра.

Решение
Задача 23E
Codeforces 23 Round 23 Beta 2010.07.09

Задача решается с помощью динамического программирования на дереве. Мы рассматриваем каждую вершину как корень поддерева и вычисляем для неё два типа состояний.

dp[v][k] — максимальное произведение размеров компонент в поддереве вершины v, при условии что компонента, содержащая вершину v, имеет размер k.

Особое состояние dp[v][0] означает, что компонента с вершиной v уже "отрезана" от родителя и её размер учтён как множитель.

Объединение с ребёнком: Если мы не отрезаем ребро к ребёнку, то мы объединяем компоненту вершины v с компонентой ребёнка, суммируя их размеры.

Отрезание ребра: Мы можем отрезать ребро к родителю, тогда текущая компонента становится отдельным множителем в произведении.

Так как произведение может быть огромным (до 700^700), используется класс BigInt с системой счисления по основанию 10000.

Сложность:

Время: O(n²) для каждой вершины, всего O(n²) операций

Память: O(n²) для хранения состояний DP

// Класс для работы с большими числами (произведение может быть огромным)
// Хранит число в системе счисления с основанием 10000
struct BigInt {
  static const int BASE = 10000;
  static const int DIGITS = 4;  // количество цифр в одном разряде

  vector digits;  // цифры в системе счисления с основанием 10000

  BigInt(int value = 0) {
    if (value == 0) {
      digits.push_back(0);
    } else {
      while (value > 0) {
        digits.push_back(value % BASE);
        value /= BASE;
      }
    }
  }

  // Умножение двух больших чисел
  BigInt operator*(const BigInt& other) const {
    BigInt result;
    result.digits.assign(digits.size() + other.digits.size(), 0);

    // Умножаем как в столбик
    for (size_t i = 0; i < digits.size(); i++) {
      for (size_t j = 0; j < other.digits.size(); j++) {
        result.digits[i + j] += digits[i] * other.digits[j];
      }
    }

    // Переносы в системе счисления
    for (size_t i = 0; i < result.digits.size() - 1; i++) {
      result.digits[i + 1] += result.digits[i] / BASE;
      result.digits[i] %= BASE;
    }

    // Удаляем ведущие нули
    while (result.digits.size() > 1 && result.digits.back() == 0) {
      result.digits.pop_back();
    }

    return result;
  }

  // Сравнение двух больших чисел
  bool operator<(const BigInt& other) const {
    if (digits.size() != other.digits.size()) {
      return digits.size() < other.digits.size();
    }
    for (int i = digits.size() - 1; i >= 0; i--) {
      if (digits[i] != other.digits[i]) {
        return digits[i] < other.digits[i];
      }
    }
    return false;  // числа равны
  }

  // Вывод числа (ручной формат без setw/setfill)
  void print() const {
    cout << digits.back();
    for (int i = digits.size() - 2; i >= 0; i--) {
      // Выводим каждую цифру с ведущими нулями до 4 знаков
      int num = digits[i];
      if (num < 1000) cout << '0';
      if (num < 100) cout << '0';
      if (num < 10) cout << '0';
      cout << num;
    }
    cout << '\n';
  }
};

const int MAXN = 705;
int n;
vector tree[MAXN];  // список смежности дерева
int subtreeSize[MAXN];   // размер поддерева
BigInt dp[MAXN][MAXN];   // dp[v][k] - максимальное произведение для поддерева v,
// где компонента, содержащая v, имеет размер k

void dfs(int vertex, int parent) {
  subtreeSize[vertex] = 1;
  dp[vertex][1] = BigInt(1);  // изначально одна компонента размера 1

  // Перебираем всех детей
  for (int child : tree[vertex]) {
    if (child == parent) continue;

    dfs(child, vertex);

    // Объединяем поддеревья - это классический рюкзак на дереве
    for (int currentSize = subtreeSize[vertex]; currentSize >= 1; currentSize--) {
      for (int childSize = subtreeSize[child]; childSize >= 0; childSize--) {
        // Если мы не отрезаем ребро к ребенку, то объединяем компоненты
        if (currentSize + childSize <= n) {
          if (dp[vertex][currentSize + childSize] <
            dp[vertex][currentSize] * dp[child][childSize]) {
            dp[vertex][currentSize + childSize] =
              dp[vertex][currentSize] * dp[child][childSize];
          }
        }
      }
    }

    subtreeSize[vertex] += subtreeSize[child];
  }

  // Важный шаг: мы можем отрезать ребро к родителю,
  // тогда компонента с вершиной vertex идёт как отдельный множитель
  for (int i = 1; i <= subtreeSize[vertex]; i++) {
    if (dp[vertex][0] < dp[vertex][i] * BigInt(i)) {
      dp[vertex][0] = dp[vertex][i] * BigInt(i);
    }
  }
}

int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);

  cin >> n;

  // Читаем дерево
  for (int i = 0; i < n - 1; i++) {
    int a, b;
    cin >> a >> b;
    tree[a].push_back(b);
    tree[b].push_back(a);
  }

  // Запускаем DFS из корня
  dfs(1, 0);

  // Ответ хранится в dp[1][0] - компонента, содержащая корень,
  // уже отделена от остального дерева
  dp[1][0].print();

  return 0;
}
