Задана карта местности, представляющая собой прямоугольную матрицу n x m, в каждой клетке которой записана средняя высота соответствующего участка местности. Прорабу Пете нужно расположить на карте несколько городов, каждый из которых на карте займет прямоугольник размера a x b клеток. Но прежде чем построить город в определенном месте, Пете необходимо убрать лишнюю землю с участка, на котором будет строиться город. Для этого он выбирает клетку наименьшей высоты на этом участке и делает высоту всех клеток на участке равной высоте этой клетки. Будем считать, что для того, чтобы опустить уровень земли с h2 до h1 (h1 ≤ h2), необходимо вывезти h2 - h1 единиц земли.

Назовем положение участка под город оптимальным, если количество вывозимой с него земли наименьшее из всех возможных положений. Петя строит города в соответствии со следующим алгоритмом: из всех оптимальных положений участка он выбирает самое верхнее. Если таких положений несколько, он выбирает среди них самое левое. На этом участке он строит город. Петя повторяет этот процесс до тех пор, пока он может построить хотя бы еще один город. На уже застроенных клетках, разумеется, строить вновь нельзя. Помогите Пете расположить города в соответствии с алгоритмом.

Входные данные

В первой строке перечислено через пробел четыре целых числа — размеры карты n, m и размеры города a, b (1 ≤ a ≤ n ≤ 1000, 1 ≤ b ≤ m ≤ 1000). Далее идут n строк по m неотрицательных целых чисел в каждой, задающие матрицу высот. Все числа не превышают 109. Числа в строках разделяются пробелами.

Выходные данные

В первой строке выведите k — число построенных городов. В следующих k строках выведите по 3 числа, разделенных пробелами, — номер строки и номер столбца левого верхнего угла очередного участка, а также объем земли, который требуется с него вывезти. Участки выводите в порядке их построения.

using namespace std;

int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);

  int n, m, a, b;
  cin >> n >> m >> a >> b;

  // Исходная матрица (1-indexed для удобства сумм)
  vector> A(n + 1, vector(m + 1, 0));
  vector> prefixSum(n + 1, vector(m + 1, 0));

  for (int i = 1; i <= n; ++i) {
    for (int j = 1; j <= m; ++j) {
      cin >> A[i][j];
      prefixSum[i][j] = A[i][j] + prefixSum[i - 1][j] +
        prefixSum[i][j - 1] - prefixSum[i - 1][j - 1];
    }
  }

  // Шаг 1: минимум в окне по строкам (ширина b)
  vector> rowMin(n + 1, vector(m - b + 2, 0));

  for (int i = 1; i <= n; ++i) {
    deque dq;
    for (int j = 1; j <= m; ++j) {
      while (!dq.empty() && A[i][dq.back()] > A[i][j]) {
        dq.pop_back();
      }
      while (!dq.empty() && dq.front() <= j - b) {
        dq.pop_front();
      }
      dq.push_back(j);
      if (j >= b) {
        rowMin[i][j - b + 1] = A[i][dq.front()];
      }
    }
  }

  // Шаг 2: минимум в окне по столбцам (высота a) от rowMin
  vector> minInRect(n - a + 2, vector(m - b + 2, 0));

  for (int j = 1; j <= m - b + 1; ++j) {
    deque dq;
    for (int i = 1; i <= n; ++i) {
      while (!dq.empty() && rowMin[dq.back()][j] > rowMin[i][j]) {
        dq.pop_back();
      }
      while (!dq.empty() && dq.front() <= i - a) {
        dq.pop_front();
      }
      dq.push_back(i);
      if (i >= a) {
        minInRect[i - a + 1][j] = rowMin[dq.front()][j];
      }
    }
  }

  // Шаг 3: вычисление ценности каждого прямоугольника
  vector> value(n - a + 2, vector(m - b + 2, 0));

  for (int i = 1; i <= n - a + 1; ++i) {
    for (int j = 1; j <= m - b + 1; ++j) {
      long long sum = prefixSum[i + a - 1][j + b - 1] -
        prefixSum[i - 1][j + b - 1] -
        prefixSum[i + a - 1][j - 1] +
        prefixSum[i - 1][j - 1];
      value[i][j] = sum - static_cast(minInRect[i][j]) * a * b;
    }
  }

  // Шаг 4: сбор координат всех прямоугольников
  vector> rectangles;
  for (int i = 1; i <= n - a + 1; ++i) {
    for (int j = 1; j <= m - b + 1; ++j) {
      rectangles.emplace_back(i, j);
    }
  }

  // Сортировка по ценности (и по координатам при равенстве)
  sort(rectangles.begin(), rectangles.end(),
    [&](const pair& p1, const pair& p2) {
      if (value[p1.first][p1.second] != value[p2.first][p2.second]) {
        return value[p1.first][p1.second] < value[p2.first][p2.second];
      }
      return p1 < p2;
    });

  // Шаг 5: выбор непересекающихся прямоугольников
  vector> used(n + 1, vector(m + 1, false));
  vector> answer;

  for (const auto& [x, y] : rectangles) {
    if (!used[x][y] && !used[x + a - 1][y] &&
      !used[x][y + b - 1] && !used[x + a - 1][y + b - 1]) {

      // Занять область
      for (int i = 0; i < a; ++i) {
        for (int j = 0; j < b; ++j) {
          used[x + i][y + j] = true;
        }
      }
      answer.emplace_back(x, y, value[x][y]);
    }
  }

  // Вывод результата
  cout << answer.size() << '\n';
  for (const auto& [x, y, val] : answer) {
    cout << x << ' ' << y << ' ' << val << '\n';
  }

  return 0;
}
