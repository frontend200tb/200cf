(Время 1с. Память 256мб)

Строка s длины n называется k-палиндромом, если она сама является палиндромом, а ее префикс и суффикс длины n / 2 являются (k - 1)-палиндромами. 0-палиндромом является любая строка (даже пустая).

Назовем палиндромностью строки s такое максимальное число k, для которого s является k-палиндромом. Например, палиндромность строки «abaaba» равна 3.

Дана строка. Ваша задача — найти сумму палиндромностей всех ее префиксов.

Входные данные

В первой строке входных данных записана непустая строка, состоящая из букв латинского алфавита и цифр. Длина строки не превосходит 5·10⁶. Регистр букв следует различать.

Выходные данные

Выведите единственное число — сумму палиндромностей всех префиксов данной строки.

Примеры

Пример 1

a2A 1

Пример 2

abacaba 6
Решение
Задача 7D
Codeforces Beta Round 7 2010.04.01

Палиндромность строки — это максимальное k, для которого строка является k-палиндромом:

0-палиндром: любая строка

1-палиндром: обычный палиндром

2-палиндром: палиндром, у которого префикс и суффикс длины ⌊n/2⌋ — 1-палиндромы

Алгоритм Манакера находит для каждой позиции радиус наибольшего палиндрома:

d[i][0] — радиус четного палиндрома с центром между i-1 и i

d[i][1] — радиус нечетного палиндрома с центром в i

Функция good(i) Проверяет, является ли префикс длины i+1 палиндромом:

Для нечетной длины (i четное): центр в i/2, радиус должен быть (i+1)/2

Для четной длины (i нечетное): центр между i/2 и i/2+1, радиус должен быть (i+1)/2

Если строка не палиндром → палиндромность = 0

Если строка палиндром → палиндромность = 1 + палиндромность её половины

#include <cstdio>
#include <cstring>
#include <algorithm>

using namespace std;

const int MAXN = 5000001;
char s[MAXN];
int d1[MAXN], d2[MAXN];
int res[MAXN];

int main() {
  fgets(s, MAXN, stdin);
  int n = strlen(s);
  if (s[n - 1] == '\n') s[n - 1] = '\0';
  n = strlen(s);

  // Манакер для нечетных
  int l = 0, r = -1;
  for (int i = 0; i < n; ++i) {
    int k = (i > r) ? 1 : min(d1[l + r - i], r - i + 1);
    while (i - k >= 0 && i + k < n && s[i - k] == s[i + k]) {
      ++k;
    }
    d1[i] = k--;
    if (i + k > r) {
      l = i - k;
      r = i + k;
    }
  }

  // Манакер для четных
  l = 0, r = -1;
  for (int i = 0; i < n; ++i) {
    int k = (i > r) ? 0 : min(d2[l + r - i + 1], r - i + 1);
    while (i - k - 1 >= 0 && i + k < n && s[i - k - 1] == s[i + k]) {
      ++k;
    }
    d2[i] = k--;
    if (i + k > r) {
      l = i - k - 1;
      r = i + k;
    }
  }

  long long ans = 0;
  for (int i = 1; i <= n; ++i) {
    bool isPal;
    if (i & 1) {
      isPal = (d1[i / 2] * 2 - 1 >= i);
    } else {
      isPal = (d2[i / 2] * 2 >= i);
    }

    if (isPal) {
      res[i] = res[i / 2] + 1;
    } else {
      res[i] = 0;
    }
    ans += res[i];
  }

  printf("%lld\n", ans);
  return 0;
}
