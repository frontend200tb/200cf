Задача B. Татарское телешоу

#include 

using namespace std;

int main() {
  int t; // число тестов
  cin >> t;

  int n; // длина строки
  int k; // число k
  string s; // бинарная строка

  while (t--) {
    cin >> n >> k >> s;

    bool flag = 1;

    // нужно сделать строку полностью нулевой
    // инвертируя попарно два символа
    // это можно сделать только если число 1 четное
    for (int r = 0; r < k; r++) {
      int cnt1 = 0; // число единиц

      for (int i = r; i < n; i += k) {
        if (s[i] == '1') {
          cnt1++;
        }
      }

      if (cnt1 % 2 != 0) {
        flag = 0;
      }
    }

    if (flag) {
      cout << "YES\n";
    } else {
      cout << "NO\n";
    }

  }
}

#include

using namespace std;

int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);

  int t;
  cin >> t;

  while (t--) {
    long long a, b, x;
    cin >> a >> b >> x;

    long long ans = abs(a - b); // только инкременты

    // Перебираем деления a и b
    long long cur_a = a;
    for (int steps_a = 0; steps_a <= 60; steps_a++) {
      long long cur_b = b;
      for (int steps_b = 0; steps_b <= 60; steps_b++) {
        long long ops = steps_a + steps_b + abs(cur_a - cur_b);
        if (ops < ans) {
          ans = ops;
        }
        // Делим cur_b, если возможно, иначе break
        if (cur_b == 0) break;
        cur_b /= x;
      }
      // Делим cur_a
      if (cur_a == 0) break;
      cur_a /= x;
    }

    cout << ans << "\n";
  }

  return 0;
}

#include 
#include 
#include 
#include 

using namespace std;

int main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);

  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector a(n);
    for (int i = 0; i < n; ++i) {
      cin >> a[i];
      --a[i];
    }

    vector> good(n, vector(n, false));
    vector> minVal(n, vector(n, 0));
    vector> maxVal(n, vector(n, 0));

    // Precompute good segments
    for (int i = 0; i < n; ++i) {
      vector freq(n, 0);
      int mn = a[i], mx = a[i];
      bool ok = true;
      for (int j = i; j < n; ++j) {
        if (++freq[a[j]] > 1) {
          ok = false;
        }
        mn = min(mn, a[j]);
        mx = max(mx, a[j]);
        if (ok && mx - mn == j - i) {
          good[i][j] = true;
        }
        minVal[i][j] = mn;
        maxVal[i][j] = mx;
      }
    }

    int ans = 0;

    // Try all possible lengths k
    for (int k = n / 2; k >= 1; --k) {
      bool found = false;
      for (int i = 0; i + k <= n && !found; ++i) {
        if (!good[i][i + k - 1]) continue;
        for (int j = i + k; j + k <= n && !found; ++j) {
          if (!good[j][j + k - 1]) continue;
          // Check if they are sequential intervals
          int m1 = minVal[i][i + k - 1];
          int m2 = minVal[j][j + k - 1];
          if (abs(m1 - m2) == k) {
            found = true;
            ans = k;
          }
        }
      }
      if (found) break;
    }

    cout << ans << '\n';
  }

  return 0;
}

Задача F1. Выборы в Саранске (простая версия)

#include

using namespace std;

const int MOD = 1e9 + 7;
const int MAXA = 5e5 + 5;

int spf[MAXA]; // smallest prime factor

void sieve() {
  for (int i = 2; i < MAXA; i++) spf[i] = i;
  for (int i = 2; i * i < MAXA; i++) {
    if (spf[i] == i) {
      for (int j = i * i; j < MAXA; j += i) {
        if (spf[j] == j) spf[j] = i;
      }
    }
  }
}

int main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);

  sieve();

  int t;
  cin >> t;
  while (t--) {
    int n, x;
    cin >> n >> x; // x = 1
    vector a(n);
    for (int i = 0; i < n; i++) cin >> a[i];

    map prime_ways;

    for (int i = 0; i < n; i++) {
      int val = a[i];
      while (val > 1) {
        int p = spf[val];
        int cnt = 0;
        while (val % p == 0) {
          val /= p;
          cnt++;
        }
        prime_ways[p] += cnt; // sum v_q(a_i)
      }
    }

    long long ans = 1;
    for (auto& [prime, sum_v] : prime_ways) {
      // ways = 1 (not using prime) + sum_v
      ans = (ans * (sum_v + 1)) % MOD;
    }

    cout << ans << '\n';
  }

  return 0;
}
