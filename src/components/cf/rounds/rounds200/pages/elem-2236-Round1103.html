Задача B. Татарское телешоу

#include 

using namespace std;

int main() {
  int t; // число тестов
  cin >> t;

  int n; // длина строки
  int k; // число k
  string s; // бинарная строка

  while (t--) {
    cin >> n >> k >> s;

    bool flag = 1;

    // нужно сделать строку полностью нулевой
    // инвертируя попарно два символа
    // это можно сделать только если число 1 четное
    for (int r = 0; r < k; r++) {
      int cnt1 = 0; // число единиц

      for (int i = r; i < n; i += k) {
        if (s[i] == '1') {
          cnt1++;
        }
      }

      if (cnt1 % 2 != 0) {
        flag = 0;
      }
    }

    if (flag) {
      cout << "YES\n";
    } else {
      cout << "NO\n";
    }

  }
}

#include

#include 

using namespace std;

int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);

  int t;
  cin >> t;

  while (t--) {
    long long a, b, x;
    cin >> a >> b >> x;

    long long ans = abs(a - b); // только инкременты

    // Перебираем деления a и b
    long long cur_a = a;
    for (int steps_a = 0; steps_a <= 60; steps_a++) {
      long long cur_b = b;
      for (int steps_b = 0; steps_b <= 60; steps_b++) {
        long long ops = steps_a + steps_b + abs(cur_a - cur_b);
        if (ops < ans) {
          ans = ops;
        }
        // Делим cur_b, если возможно, иначе break
        if (cur_b == 0) break;
        cur_b /= x;
      }
      // Делим cur_a
      if (cur_a == 0) break;
      cur_a /= x;
    }

    cout << ans << "\n";
  }

  return 0;
}

#include 

using namespace std;

void solve();

int main() {
  int t; cin >> t;
  while (t--) solve();
  return 0;
}

void solve() {
  int n, k; cin >> n >> k;
  vector A(n);
  for (auto& x : A) cin >> x;
  sort(A.begin(), A.end());
  vector> a = { {A[0], 1} };
  for (int i = 1; i < n; ++i) {
    if (A[i] == A[i - 1]) a.back().second++;
    else a.emplace_back(A[i], 1);
  }

  while (a.size() > 0) {
    n = a.size();
    if (a[n - 1].second % 2 == 0) {
      cout << "YES\n";
      return;
    }
    if (n == 1) {
      cout << "NO\n";
      return;
    }
    if (a[n - 1].first - a[n - 2].first <= k) {
      cout << "YES\n";
      return;
    }
    a.pop_back();
  }
}

Задача E. Дружеские подарки

#include 
#include 
#include 

using namespace std;

void solve();

int main() {
  int t;
  cin >> t;
  while (t--) {
    solve();
  }

  return 0;
}

void solve() {
  int n; cin >> n;
  vector a(n);
  for (auto& x : a) cin >> x, --x;
  vector> able(n, vector(n));
  for (int i = 0; i < n; ++i) {
    vector us(n);
    int mn = a[i], mx = a[i];
    for (int j = i; j < n; ++j) {
      if (us[a[j]]) break;
      us[a[j]] = 1;
      mn = min(mn, a[j]), mx = max(mx, a[j]);
      if (mx - mn == j - i) able[mn][mx] = 1;
    }
  }

  for (int ans = n; ans > 0; --ans) {
    for (int i = 0; i + 2 * ans <= n; ++i) {
      if (able[i][i + ans - 1] && able[i + ans][i + 2 * ans - 1]) {
        cout << ans << '\n';
        return;
      }
    }
  }
  cout << "0\n";
}

Задача F1. Выборы в Саранске (простая версия)

#include

using namespace std;

const int MOD = 1e9 + 7;
const int MAXA = 5e5 + 5;

int spf[MAXA]; // smallest prime factor

void sieve() {
  for (int i = 2; i < MAXA; i++) spf[i] = i;
  for (int i = 2; i * i < MAXA; i++) {
    if (spf[i] == i) {
      for (int j = i * i; j < MAXA; j += i) {
        if (spf[j] == j) spf[j] = i;
      }
    }
  }
}

int main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);

  sieve();

  int t;
  cin >> t;
  while (t--) {
    int n, x;
    cin >> n >> x; // x = 1
    vector a(n);
    for (int i = 0; i < n; i++) cin >> a[i];

    map prime_ways;

    for (int i = 0; i < n; i++) {
      int val = a[i];
      while (val > 1) {
        int p = spf[val];
        int cnt = 0;
        while (val % p == 0) {
          val /= p;
          cnt++;
        }
        prime_ways[p] += cnt; // sum v_q(a_i)
      }
    }

    long long ans = 1;
    for (auto& [prime, sum_v] : prime_ways) {
      // ways = 1 (not using prime) + sum_v
      ans = (ans * (sum_v + 1)) % MOD;
    }

    cout << ans << '\n';
  }

  return 0;
}
