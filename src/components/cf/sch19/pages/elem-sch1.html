#include <iostream>
#include <vector>

using namespace std;

int main() {
  // ускорение ввода-вывода
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);
  cout.tie(nullptr);

  // ввод данных
  int n; // число остановок
  cin >> n;
  vector<int> A(n + 1);
  vector<int> B(n + 1);
  for (int i = 1; i <= n; i++) {
    cin >> A[i] >> B[i];
  }

  // решение
  int cap = 0;
  int capm = 0;

  for (int i = 1; i <= n; i++) {
    cap -= A[i];
    cap += B[i];
    if (cap > capm) {
      capm = cap;
    }
  }

  // вывод результата
  cout << capm;
}

#include <iostream>

using namespace std;

int main() {
  // ввод данных
  int n; // длина массива
  cin >> n;
  int in;

  // соберем частотный массив
  int F[4] = { 0,0,0,0 };
  for (int i = 0; i < n; i++) {
    cin >> in;
    F[in]++;
  }

  // решение
  int res = 0;
  // находим наибольшее из трех чисел
  if (F[1] >= F[2] && F[1] >= F[3]) {
    res = F[2] + F[3];
  } else if (F[2] >= F[1] && F[2] >= F[3]) {
    res = F[1] + F[3];
  } else {
    res = F[1] + F[2];
  }

  // вывод результата
  cout << res;
}
