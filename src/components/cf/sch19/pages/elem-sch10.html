Задача C. Плюс один на подмножестве

Решение 
#include <iostream>

using namespace std;

int main() {
  int t; // число тестов
  cin >> t;

  while (t--) {
    int n; // длина массива
    cin >> n;

    int first; // первый элемент массива
    cin >> first;
    int mina = first;
    int maxa = first;
    int in = 0; // следующий элемент массива

    for (int i = 1; i < n; i++) {
      cin >> in;
      if (mina > in) {
        mina = in;
      }
      if (in > maxa) {
        maxa = in;
      }
    }

    cout << maxa - mina << '\n';
  }
}
