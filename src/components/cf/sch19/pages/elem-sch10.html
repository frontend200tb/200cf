Задача A. Следующий раунд

Решение 
#include <iostream>

using namespace std;

int main() {
  int n; // число участников
  int k; // число баллов
  cin >> n >> k;

  int cnt = 0; // число участников в след. раунде
  int bk; // число баллов k-го места

  for (int i = 1; i <= n; i++) {
    int b; // число баллов
    cin >> b;
    if (b > 0) {
      if (i < k) {
        cnt++;
      }
      if (i == k) {
        bk = b;
        cnt++;
      }
      if (i > k && b == bk) {
        cnt++;
      }
    }
  }

  cout << cnt;
}

Задача C. Плюс один на подмножестве

Решение 
#include <iostream>

using namespace std;

int main() {
  int t; // число тестов
  cin >> t;

  while (t--) {
    int n; // длина массива
    cin >> n;

    int first; // первый элемент массива
    cin >> first;
    int mina = first;
    int maxa = first;
    int in = 0; // следующий элемент массива

    for (int i = 1; i < n; i++) {
      cin >> in;
      if (mina > in) {
        mina = in;
      }
      if (in > maxa) {
        maxa = in;
      }
    }

    cout << maxa - mina << '\n';
  }
}
