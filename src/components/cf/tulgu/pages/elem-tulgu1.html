using namespace std;

  int x; // фиксированная оплата

Задача B. Настоящие друзья

Решение

Задача на бинарный поиск по ответу. Нужно найти такую коробку, чтобы число всех конфет сверху до этой коробки было равно числу конфет в нижних коробках.

Сумма чисел от 1 до n

n * (n+1) / 2

Сумма чисел от A до B

(B-A+1) * (A+B) / 2 
#include <iostream>

using namespace std;

typedef long long ll;

ll sum1(ll n) {
  return n * (n + 1) / 2;
}

ll sum2(ll A, ll B) {
  return (B - A + 1) * (A + B) / 2;
}

int main() {
  int t; // число тестов
  cin >> t;

  while (t--) {
    ll k; // число коробок
    cin >> k;

    // сумма чисел от 1 до n
    // n * (n+1) / 2
    // сумма чисел от A до B
    // (B-A+1) * (A+B) / 2

    ll left = 0;
    ll right = k + 1;
    ll mid;

    // бинарный поиск по ответу
    while (right - left > 1) {
      mid = (right + left) / 2;
      if (sum1(mid) <= sum2(mid+1, k)) {
        left = mid;
      } else {
        right = mid;
      }
    }

    if (sum1(left) == sum2(left+1, k)) {
      cout << "yes" << '\n';
    } else {
      cout << "no" << '\n';
    }
  }
}

Задача C. Пирамида

Решение

Создадим вектор кирпичей, для каждого кирпича будем хранить только его меньшую сторону. Отсортируем вектор кирпичей по размеру. Пройдем по всем кирпичам и если очередной кирпич может сделать еще один слой пирамиды, то будем увеличивать высоту пирамиды на один.

#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n; // число кирпичей
  cin >> n;
  int a; // меньшая сторона кирпича
  int b; // большая стороная кирпича
  vector<int> A(n);
  for (int i = 0; i < n; i++) {
    cin >> a >> b; // стороны кирпича
    if (a > b) {
      A[i] = b; // размер кирпича
    } else {
      A[i] = a; // размер кирпича
    }
  }

  sort(A.begin(), A.end());
  int k = 0; // высота пирамиды

  for (int i = 0; i < n; i++) {
    if (A[i] >= 2 * (k + 1) - 1) {
      k++;
    }
  }

  cout << k;
}

Задача D. Простой феникс

Решение

эффективная атака возможна, если существуют два врага, расположенные рядом по горизонтали или вертикали, и с одной из сторон от этой пары есть свободная клетка для приземления феникса

Храним позиции врагов в set для быстрой проверки и в vector для итерации

Для каждого врага проверяем, есть ли соседний враг справа или снизу (чтобы не проверять пары дважды).

Для каждой найденной пары проверяем, есть ли свободная клетка для приземления феникса с любой стороны от пары.

#include <iostream>
#include <vector>
#include <set>

using namespace std;

typedef long long ll;

int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);

  ll n, m; // размер поля
  ll k; // число врагов
  cin >> n >> m >> k;

  set<pair<ll, ll> > enemies;
  vector<pair<ll, ll> > enemy_positions(k);

  for (int i = 0; i < k; i++) {
    cin >> enemy_positions[i].first >> enemy_positions[i].second;
    enemies.insert(enemy_positions[i]);
  }

  // Проверяем каждую пару соседних врагов
  for (const auto& pos : enemy_positions) {
    ll x = pos.first;
    ll y = pos.second;

    // Горизонтальная пара: (x, y) и (x, y+1)
    if (enemies.count({ x, y + 1 })) {
      // Проверяем возможность атаки слева от пары
      if (y - 1 >= 1 && !enemies.count({ x, y - 1 })) {
        cout << "yes";
        return 0;
      }
      // Проверяем возможность атаки справа от пары
      if (y + 2 <= m && !enemies.count({ x, y + 2 })) {
        cout << "yes";
        return 0;
      }
    }

    // Вертикальная пара: (x, y) и (x+1, y)
    if (enemies.count({ x + 1, y })) {
      // Проверяем возможность атаки сверху от пары
      if (x - 1 >= 1 && !enemies.count({ x - 1, y })) {
        cout << "yes";
        return 0;
      }
      // Проверяем возможность атаки снизу от пары
      if (x + 2 <= n && !enemies.count({ x + 2, y })) {
        cout << "yes";
        return 0;
      }
    }
  }

  cout << "no";
  return 0;
}

#include <iostream>

#include <vector>

using namespace std;

int main() {
    string s;
    cin >> s;
    int n = s.size();

    if (n < 3) {
        cout << 0 << endl;
        return 0;
    }

    // dp[i][r1][r2] - количество способов разбить префикс длины i на 2 числа,
    // чтобы остаток от деления первого на 2 был r1, а остаток от деления суммы цифр второго на 3 был r2
    // третье число будет формироваться из оставшейся части строки (после второго разбиения)

    vector<vector<vector<long long> > > dp(n + 1,
        vector<vector<long long> >(2,
            vector<long long>(3, 0)));

    // Инициализация: до первого разбиения нет чисел
    dp[0][0][0] = 1;

    for (int i = 0; i < n; ++i) {
        int digit = s[i] - '0';
        for (int r1 = 0; r1 < 2; ++r1) {
            for (int r2 = 0; r2 < 3; ++r2) {
                if (dp[i][r1][r2] == 0) continue;

                // Продолжаем первое число
                if (i > 0) { // первое число должно быть натуральным, т.е. начинаться не с нуля или быть непустым
                    int new_r1 = (r1 * 10 + digit) % 2;
                    dp[i + 1][new_r1][r2] += dp[i][r1][r2];
                }

                // Начинаем второе число с текущей цифры
                // Второе число не может начинаться с нуля, если оно не однозначное?
                // Оно должно быть натуральным, значит, если digit == 0, то это допустимо только если это единственная цифра второго числа?
                // Нет, число может быть "0"? Нет, натуральное число не может начинаться с нуля вообще.
                // Поэтому если digit == 0, то мы не можем начать второе число с этой цифры.
                // Но если мы продолжаем второе число, то внутри него нули допустимы.
                // У нас r2 - это остаток суммы цифр второго числа по модулю 3.
                // Мы начинаем второе число с текущей цифры, если до этого у нас уже было первое число (т.е. i >= 1 и мы уже закончили первое число)
                // Но в нашей DP мы храним r1 и r2 для первых двух чисел, третье — остаток строки.
                // Пересмотрим подход: будем перебирать позиции разбиений.

                // Лучше перебрать позиции первых двух разбиений.
            }
        }
    }

    // Вместо DP с состояниями можно перебрать позиции разбиений i и j (1 <= i < j < n)
    // Первое число: s[0..i-1], второе: s[i..j-1], третье: s[j..n-1]
    // Все без ведущих нулей и с соответствием делимости.

    long long ans = 0;

    for (int i = 1; i <= n - 2; ++i) {
        // Первое число s[0..i-1]
        if (s[0] == '0') continue; // ведущий ноль в первом числе
        string first = s.substr(0, i);
        // Проверка делимости на 2
        if ((first.back() - '0') % 2 != 0) continue; // делимость на 2 определяется последней цифрой

        for (int j = i + 1; j <= n - 1; ++j) {
            // Второе число s[i..j-1]
            if (s[i] == '0') continue; // ведущий ноль во втором числе
            // Делимость на 3 — сумма цифр делится на 3
            int sum2 = 0;
            for (int k = i; k < j; ++k) sum2 += (s[k] - '0');
            if (sum2 % 3 != 0) continue;

            // Третье число s[j..n-1]
            if (s[j] == '0') continue; // ведущий ноль в третьем числе
            string third = s.substr(j);
            // Делимость на 4: если длина >= 2, то по последним двум цифрам, иначе по последней
            int last_two;
            if (third.size() >= 2) {
                last_two = (third[third.size() - 2] - '0') * 10 + (third.back() - '0');
            } else {
                last_two = third.back() - '0';
            }
            if (last_two % 4 != 0) continue;

            ++ans;
        }
    }

    cout << ans;
