Задача A. Ограбление

Решение 
#include <iostream>

using namespace std;

int main() {
  int a; // позиция преступника
  int b, c; // позиции сигнализации
  int n; // число купюр
  cin >> a >> b >> c >> n;

  int in; // следующий сейф
  int cnt = 0; // число купюр между сейфами b и c

  // соберем купюры не доходя до сигнализаций
  while (n--) {
    cin >> in;
    if (in > b && in < c) {
      cnt++;
    }
  }

  cout << cnt;
}

Задача C. Стол-конструктор

Решение 
#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n; // число ножек у стола
  cin >> n;

  int d = 2 * n; // число деталей
  long long sum = 0; // сумма длин всех деталей
  vector<int> A(d);
  for (int i = 0; i < d; i++) {
    cin >> A[i];
    sum += A[i];
  }

  // найдем длину одной ножки
  long long len = sum / n;

  vector<pair<int, int> > P; // пары ножек
  bool ispair = false; // пара не найдена
  for (int i = 0; i < d; i++) {
    ispair = false;
    for (int j = 0; j < P.size(); j++) {
      if (P[j].first + A[i] == len && !P[j].second && !ispair) {
        P[j].second = A[i];
        ispair = true;
      }
    }
    if (!ispair) {
      P.push_back(make_pair( A[i], 0 ));
    }
  }

  for (int i = 0; i < P.size(); i++) {
    cout << P[i].first << ' ' << P[i].second << '\n';
  }
}

Задача G. Автосалон

Решение 
#include <iostream>
#include <vector>
#include <set>
#include <map>

using namespace std;

int main() {
  int n; // число машин
  int k; // число различных чисел
  cin >> n >> k;

  set<int> S;
  vector<int> A(n + 1);
  for (int i = 1; i <= n; i++) {
    cin >> A[i];
    S.insert(A[i]);
  }

  if (S.size() < k) {
    cout << "-1 -1";
  } else {
    // метод двух указателей
    map<int, int> M;
    int l = 1; // левый указатель
    int r = 1; // правый указатель

    // находим правую границу
    while (M.size() < k) {
      if (!M[A[r]]) {
        M[A[r]] = 0;
      }
      M[A[r]]++;
      r++;
    }

    // находим левую границу
    while (M.size() == k) {
      M[A[l]]--;
      if (M[A[l]] == 0) {
        M.erase(A[l]);
      }
      l++;
    }

    cout << l - 1 << ' ' << r - 1;
  }

}
