Задача A. Ограбление

Решение 
#include <iostream>

using namespace std;

int main() {
  int a; // позиция преступника
  int b, c; // позиции сигнализации
  int n; // число купюр
  cin >> a >> b >> c >> n;

  int in; // следующий сейф
  int cnt = 0; // число купюр между сейфами b и c

  // соберем купюры не доходя до сигнализаций
  while (n--) {
    cin >> in;
    if (in > b && in < c) {
      cnt++;
    }
  }

  cout << cnt;
}

Задача C. Стол-конструктор

Решение 
#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n; // число ножек у стола
  cin >> n;

  int d = 2 * n; // число деталей
  long long sum = 0; // сумма длин всех деталей
  vector<int> A(d);
  for (int i = 0; i < d; i++) {
    cin >> A[i];
    sum += A[i];
  }

  // найдем длину одной ножки
  long long len = sum / n;

  vector<pair<int, int> > P; // пары ножек
  bool ispair = false; // пара не найдена
  for (int i = 0; i < d; i++) {
    ispair = false;
    for (int j = 0; j < P.size(); j++) {
      if (P[j].first + A[i] == len && !P[j].second && !ispair) {
        P[j].second = A[i];
        ispair = true;
      }
    }
    if (!ispair) {
      P.push_back(make_pair( A[i], 0 ));
    }
  }

  for (int i = 0; i < P.size(); i++) {
    cout << P[i].first << ' ' << P[i].second << '\n';
  }
}

Задача D. Шоколадки

Решение

Для каждой шоколадки размером n×m и нужного количества долек p мы пытаемся найти такие целые a и b, чтобы:

  • a x b = p (прямоугольник нужной площади)
  • a ≤ n и b ≤ m (прямоугольник помещается в шоколадку по вертикали и горизонтали)

Мы перебираем все возможные значения a от 1 до n и проверяем, делится ли p на a без остатка. Если делится, то b = p / a, и проверяем, помещается ли b по ширине m.

Мы также проверяем случай, когда сторона a идет по ширине (от 1 до m), а сторона b - по высоте.

#include <iostream>

using namespace std;

int main() {
  int q; // число шоколадок
  cin >> q;

  while (q--) {
    int n, m; // размер шоколадки
    int p; // число плиток которые отломать
    cin >> n >> m >> p;

    bool res = false;

    // проверим все возможные стороны прямоугольника a и b
    for (int a = 1; a <= n; a++) {
      if (p % a == 0) {
        int b = p / a;
        if (b <= m) {
          res = true;
          break;
        }
      }
    }

    if (!res) {
      for (int a = 1; a <= m; a++) {
        if (p % a == 0) {
          int b = p / a;
          if (b <= n) {
            res = true;
            break;
          }
        }
      }
    }

    if (res) {
      cout << "Yes\n";
    } else {
      cout << "No\n";
    }
  }
}

Задача F. C–-

Решение

Чтобы квадрат числа поместился в больший размер типа данных, размер должен быть в два раза больше.

#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n; // число типов в языке c--
  cin >> n;

  vector<int> A(n);
  for (int i = 0; i < n; i++) {
    cin >> A[i]; // размер типа
  }

  sort(A.begin(), A.end());
  bool isThrow = false;

  for (int i = 0; i < n - 1; i++) {
    if (A[i] * 2 > A[i + 1] && A[i+1] > A[i]) {
      isThrow = true;
    }
  }

  if (isThrow) {
    cout << "YES";
  } else {
    cout << "NO";
  }
}

Задача G. Автосалон

Решение 
#include <iostream>
#include <vector>
#include <set>
#include <map>

using namespace std;

int main() {
  int n; // число машин
  int k; // число различных чисел
  cin >> n >> k;

  set<int> S;
  vector<int> A(n + 1);
  for (int i = 1; i <= n; i++) {
    cin >> A[i];
    S.insert(A[i]);
  }

  if (S.size() < k) {
    cout << "-1 -1";
  } else {
    // метод двух указателей
    map<int, int> M;
    int l = 1; // левый указатель
    int r = 1; // правый указатель

    // находим правую границу
    while (M.size() < k) {
      if (!M[A[r]]) {
        M[A[r]] = 0;
      }
      M[A[r]]++;
      r++;
    }

    // находим левую границу
    while (M.size() == k) {
      M[A[l]]--;
      if (M[A[l]] == 0) {
        M.erase(A[l]);
      }
      l++;
    }

    cout << l - 1 << ' ' << r - 1;
  }

}
