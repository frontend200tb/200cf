Задача B. Отложенные операции

Решение

Алгоритм работает следующим образом:

  1. Сначала находим для каждого предмета индекс последнего дня, когда по нему приходит задание.
  2. Проходим по дням слева направо, храня множество активных предметов (тех, по которым уже были задания, но еще не выполненные).
  3. Когда встречаем день, который является последним для какого-то предмета, мы выполняем все накопившиеся задания по этому предмету в этот день (ставим номер предмета в результат) и очищаем множество активных предметов.
  4. В остальные дни Дима бездельничает (ставим 0).

Это решение гарантирует минимальное количество дней, занятых домашней работой, так как мы выполняем задания только тогда, когда это действительно необходимо - в последний день, когда встречается задание по данному предмету.

Временная сложность: O(n log n) из-за использования set.

Пространственная сложность: O(n + k).

#include <iostream>
#include <vector>
#include <set>

using namespace std;

int main() {
  int n; // число дней
  int k; // число предметов
  cin >> n >> k;

  vector<int> a(n);
  vector<int> last_pos(k + 1, -1); // последняя позиция для каждого предмета
  vector<bool> has_task(k + 1, false); // есть ли задания по предмету

  // Читаем входные данные и запоминаем последние позиции
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    last_pos[a[i]] = i;
    has_task[a[i]] = true;
  }

  vector<int> result(n, 0); // изначально все дни - безделье
  set<int> active_subjects; // предметы, задания по которым еще не выполнены

  for (int i = 0; i < n; i++) {
    int subject = a[i];

    // Добавляем текущий предмет в активные, если он еще там не был
    active_subjects.insert(subject);

    // Если это последнее задание по данному предмету
    if (last_pos[subject] == i) {
      // Выполняем все накопившиеся задания по этому предмету
      result[i] = subject;

      // Очищаем выполненные предметы
      active_subjects.clear();
    }
  }

  // Выводим результаты
  for (int i = 0; i < n; i++) {
    cout << result[i] << (i < n - 1 ? " " : "");
  }
}

Задача D. Артем в армии

Решение 
#include <iostream>

using namespace std;

int main() {
  int n; // число приказов
  int k; // танк Артема
  cin >> n >> k;
  int a, b; // номера танков от 1 до 3

  for (int i = 0; i < n; i++) {
    cin >> a >> b;
    if (k == a) {
      k = b;
    } else if (k == b) {
      k = a;
    }
  }

  cout << k;
}
