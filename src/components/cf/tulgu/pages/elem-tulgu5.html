Задача A. Пирамидальные пути

Решение 
#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n; // число вершин в графе
  cin >> n;

  vector<int> A(n);
  cin >> A[0]; // первая вершина
  cin >> A[1] ;// вторая вершина

  if (A[0] > A[1]) {
    bool dir = 0; // идем вниз
    int i = 2;
    while (i < n) {
      cin >> A[i];
      if (A[i] > A[i-1]) {
        dir = 1; // идем вверх
        i++;
        break;
      }
      i++;
    }
    while (i < n) {
      cin >> A[i];
      if (A[i-1] > A[i]) {
        dir = 0; // идем вниз
        i++;
        break;
      }
      i++;
    }

    if (dir == 1) {
      cout << "YES";
    } else {
      cout << "NO";
    }
  }

  if (A[1] > A[0]) {
    bool dir = 1; // идем вверх
    int i = 2;
    while (i < n) {
      cin >> A[i];
      if (A[i-1] > A[i]) {
        dir = 0; // идем вниз
        i++;
        break;
      }
      i++;
    }
    while (i < n) {
      cin >> A[i];
      if (A[i] > A[i-1]) {
        dir = 1; // идем вверх
        i++;
        break;
      }
      i++;
    }

    if (dir == 0) {
      cout << "YES";
    } else {
      cout << "NO";
    }
  }
}

}

  ld x; // нужная сила
  cin >> n >> x;

  vector<ld> A(n);
  for (int i = 0; i < n; i++) {
    cin >> A[i];
  }

  sort(A.begin(), A.end(), s);
  ld sum = 0;

  for (int i = 0; i < n; i++) {
    sum += A[i] / (i + 1);
