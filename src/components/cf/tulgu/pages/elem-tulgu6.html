Задача A. Качели

Решение

Из трех детей самый тяжелый должен сесть на одну сторону качелей, а двое других на другую сторону. Если разность масс превышает допустимую, то нужно взять камень.

#include <iostream>
#include <algorithm>

using namespace std;

int main() {
  int a, b, c; // массы детей
  int d; // допустимая разность
  cin >> a >> b >> c >> d;

  int m = max(a, max(b, c)); // самый тяжелый ребенок
  int dif = abs((a + b + c) - m - m); // разница на качелях

  if (dif > d) {
    cout << dif - d;
  } else {
    cout << 0;
  }
}
